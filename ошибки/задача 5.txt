#!/bin/bash

greet() {
    name="${1:-Гость}"
    if [ -z "$1" ]; then
        name="${1:-Незнакомец}"
    fi
    echo "Привет, $name"
}

# Демонстрация
greet "Анна"
greet ""
greet

# С проверкой через ?
greet() {
    echo "Привет, ${1:?Имя обязательно}"
}

greet "Иван"
