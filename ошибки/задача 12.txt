
#!/bin/bash

# Включаем строгий режим обработки ошибок
set -e

# Пример команды, которая завершится с ошибкой
false || true  # Используем || true для игнорирования ошибки

# Успешная команда после ошибочной
echo "Скрипт продолжает выполнение"

# Альтернативный способ с if
if ! false; then
    echo "Команда успешно выполнена"
else
    echo "Команда завершилась с ошибкой, но скрипт не прервался"
fi
# Пример с set +e (не рекомендуется использовать)
set +e
false
echo "Ошибки игнорируются"
set -e  # Важно включить обратно!
