
#!/bin/bash

required_tools=(git curl jq)
missing=()

for tool in "${required_tools[@]}"; do
    if ! command -v "$tool" &> /dev/null; then
        missing+=("$tool")
    fi
done

if [ "${#missing[@]}" -gt 0 ];
then
    echo "Отсутствуют: ${missing[*]}"
    exit 127
else
    echo "Все зависимости на месте"
fi
