#!/bin/bash while true; do read -p "Введите число для таблицы умножения (от 1 до 10):
 " num if ! [[ "$num" =~ ^[0-9]+$ ]]; 
then echo "Ошибка: введено не число.
" continue fi if (( num >= 1 && num <= 10 )); then 
echo "Таблица умножения на $num:" 
for ((i=1; i<=10; i++)); do 
echo "$num x $i = $((num * i))"
 done break else echo "Ошибка: 
число должно быть от 1 до 10." 
fi 
done
